62 // splitmix64 each element combined with the hash so far. Order matters, so keys with the same values in a
63 // different order (or the same value repeated) don't get the same hash. Too many identical hashes would make
64 // the item hash table grow until running out of memory.
65 uint64_t hash = 0;
66 for(size_t i = 0; i < _Size; i++)
67 {
68 uint64_t x = hash + (uint64_t)m_values[i] + 0x9e3779b97f4a7c15ULL;
69 x = (x ^ (x >> 30ULL)) * 0xbf58476d1ce4e5b9ULL;
70 x = (x ^ (x >> 27ULL)) * 0x94d049bb133111ebULL;
71 x = x ^ (x >> 31ULL);
72 hash = x;
73 }
